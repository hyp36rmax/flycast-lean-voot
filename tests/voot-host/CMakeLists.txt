cmake_minimum_required(VERSION 3.20)
project(VootHostRegression LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "These retained host fixtures require Windows x64.")
endif()
get_filename_component(RUNTIME_ROOT "../.." ABSOLUTE)
enable_testing()
foreach(fixture sort palette dispatch socket)
    add_executable(${fixture}-tests ${fixture}.cpp)
    target_compile_features(${fixture}-tests PRIVATE cxx_std_17)
    target_include_directories(${fixture}-tests PRIVATE "${RUNTIME_ROOT}/core/hw/pvr" "${RUNTIME_ROOT}/core/deps/xbyak")
    add_test(NAME ${fixture} COMMAND ${fixture}-tests)
endforeach()
target_link_libraries(socket-tests PRIVATE ws2_32)
