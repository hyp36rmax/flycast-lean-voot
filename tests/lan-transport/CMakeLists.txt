cmake_minimum_required(VERSION 3.20)
project(VootLanTransportTests LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "These retained transport fixtures require Windows x64.")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(lan08-fixture fixture.cpp)
target_compile_features(lan08-fixture PRIVATE cxx_std_17)
target_compile_definitions(lan08-fixture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_include_directories(lan08-fixture PRIVATE "../../core/network")
target_link_libraries(lan08-fixture PRIVATE ws2_32)
enable_testing()
add_test(NAME lan-transport COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/run_transport.py" "$<TARGET_FILE:lan08-fixture>" "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(lan-transport PROPERTIES TIMEOUT 130 RUN_SERIAL TRUE)
